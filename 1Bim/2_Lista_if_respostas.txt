//Exemplo char
#include <stdio.h>
char sexo;
char nome[100];
main(){
	printf("Digite o sexo da pessoa:\n");
	scanf("%c", &sexo);
	if(sexo == 'm'){
		printf("\nMasculino");
	}else{
		printf("\nFeminino");
	}
	printf("\nDigite um nome:\n");
	fflush(stdin);
	fgets(nome, 100, stdin);
	printf("\nNome digitado: %s ", nome);
}

#include <stdio.h>
#include <locale.h>
int quantidade;
float valorTotal;
main(){
	setlocale(LC_ALL, "Portuguese");
	printf("Digite a quantidade de maçãs:\n");	
	scanf("%i", &quantidade);
	if(quantidade >= 12){
		valorTotal = (quantidade * 1.0);		
	}else{
		valorTotal = (quantidade * 1.3);		
	}
	printf("\nValor total: %.2f", valorTotal);
}

#include <stdio.h>
int v1, v2, dif;
main(){
	printf("Digite o primeiro valor:\n");
	scanf("%i", &v1);
	printf("Digite o segundo valor:\n");
	scanf("%i", &v2);
	if(v1 > v2){
		dif = v1-v2;
	}else if(v2>v1){
		dif = v2-v1;
	}else{
		printf("\nnumeros iguais");
	}
	printf("\nDiferença: %i", dif);
}
