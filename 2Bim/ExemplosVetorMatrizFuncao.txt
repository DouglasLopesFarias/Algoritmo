//==== exemplo vetor 
#include <stdio.h>
main(){
	float notas[4];	
	for(int i=0; i<4; i++){
		printf("Digite a nota do %d bimestre:\n ", i+1);
		scanf("%f", &notas[i]);
	}
	for(int i=0; i<4; i++){
		printf("\nNota do %d bimestre: %.2f", i+1, notas[i]);		
	}	
}

//==== exemplo matriz
#include <stdio.h>
main(){
	int matriz[3][3];
	for(int i=0; i<3; i++){
		for(int j=0; j<3; j++){
			matriz[i][j] = i+j;
		}
	}	
	for(int i=0; i<3; i++){
		printf("\n");
		for(int j=0; j<3; j++){
			printf("\nLinha %d coluna %d = %d", i, j, matriz[i][j]); 
		}		
	}
}
